cmake_minimum_required(VERSION 3.16)
project(brave_webos_display C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(webos_display_protocol STATIC src/display/protocol.c)
target_include_directories(webos_display_protocol PUBLIC src/display)
target_compile_options(webos_display_protocol PRIVATE -Wall -Wextra -Werror)

option(BUILD_DISPLAY_HOST "Build the native SDL display/input host" OFF)
if(BUILD_DISPLAY_HOST)
    find_package(SDL3 REQUIRED CONFIG)
    add_executable(webos-display src/display/host.c)
    target_link_libraries(webos-display PRIVATE webos_display_protocol SDL3::SDL3)
    target_compile_options(webos-display PRIVATE -Wall -Wextra -Werror)
    add_executable(browser-exec src/display/browser-exec.c)
    target_compile_options(browser-exec PRIVATE -Wall -Wextra -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(protocol_test tests/protocol-test.c)
    target_link_libraries(protocol_test PRIVATE webos_display_protocol)
    target_compile_options(protocol_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_test(NAME display_protocol COMMAND protocol_test)
    if(BUILD_DISPLAY_HOST)
        add_executable(input_test tests/input-test.c)
        target_link_libraries(input_test PRIVATE webos_display_protocol SDL3::SDL3)
        target_compile_options(input_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        add_test(NAME native_input COMMAND input_test)
    endif()
endif()
