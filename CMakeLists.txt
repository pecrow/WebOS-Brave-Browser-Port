cmake_minimum_required(VERSION 3.16)
project(brave_webos_display C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(webos_display_protocol STATIC src/display/protocol.c)
target_include_directories(webos_display_protocol PUBLIC src/display)
target_compile_options(webos_display_protocol PRIVATE -Wall -Wextra -Werror)

add_library(webos_media_protocol STATIC src/media/webos_media_protocol.c)
target_include_directories(webos_media_protocol PUBLIC src/media)
target_compile_options(webos_media_protocol PRIVATE -Wall -Wextra -Werror)

option(BUILD_NATIVE_MEDIA "Build the ARM32 Starfish/ACB media helper" OFF)
if(BUILD_NATIVE_MEDIA)
    enable_language(CXX)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(ACBAPI REQUIRED AcbAPI)
    pkg_check_modules(STARFISH REQUIRED libplayerAPIs)
    pkg_check_modules(GLIB REQUIRED glib-2.0)
    find_path(STARFISH_INCLUDE_DIR StarfishMediaAPIs.h REQUIRED)
    add_executable(webos-media src/media/native_media.cc)
    set_target_properties(webos-media PROPERTIES
        CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    target_compile_features(webos-media PRIVATE cxx_std_11)
    target_include_directories(webos-media PRIVATE
        ${STARFISH_INCLUDE_DIR} ${ACBAPI_INCLUDE_DIRS} ${GLIB_INCLUDE_DIRS})
    target_link_directories(webos-media PRIVATE
        ${ACBAPI_LIBRARY_DIRS} ${STARFISH_LIBRARY_DIRS} ${GLIB_LIBRARY_DIRS})
    target_link_libraries(webos-media PRIVATE webos_media_protocol
        ${ACBAPI_LIBRARIES} ${STARFISH_LIBRARIES} ${GLIB_LIBRARIES} pthread)
    target_compile_options(webos-media PRIVATE -Wall -Wextra -Werror)
endif()

option(BUILD_DISPLAY_HOST "Build the native SDL display/input host" OFF)
if(BUILD_DISPLAY_HOST)
    find_package(SDL3 REQUIRED CONFIG)
    add_executable(webos-display src/display/host.c)
    target_link_libraries(webos-display PRIVATE webos_display_protocol SDL3::SDL3)
    target_compile_options(webos-display PRIVATE -Wall -Wextra -Werror)
    add_executable(browser-exec src/display/browser-exec.c)
    target_compile_options(browser-exec PRIVATE -Wall -Wextra -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(protocol_test tests/protocol-test.c)
    target_link_libraries(protocol_test PRIVATE webos_display_protocol)
    target_compile_options(protocol_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_test(NAME display_protocol COMMAND protocol_test)
    add_executable(media_protocol_test tests/media-protocol-test.c)
    target_link_libraries(media_protocol_test PRIVATE webos_media_protocol)
    target_compile_options(media_protocol_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_test(NAME media_protocol COMMAND media_protocol_test)
    if(BUILD_DISPLAY_HOST)
        add_executable(input_test tests/input-test.c)
        target_link_libraries(input_test PRIVATE webos_display_protocol SDL3::SDL3)
        target_compile_options(input_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        add_test(NAME native_input COMMAND input_test)
    endif()
endif()
